_# --- 10.100.100.3 DEVICE CONNECTION ---
_HOST: 172.29.112.112
_USERNAME: clab
_PASSWORD: ${JUMP_PASSWORD}
_PORT: 2222
_EXPECT: #\s*$
_LOGID: 10.100.100.3

_IF: continue connecting
_THEN: yes
_IF: .*assword:
_THEN: ${DEVICE_PASSWORD}

ssh -o PubkeyAuthentication=no -o PreferredAuthentications=password -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null admin@10.100.100.3

environment no more
_CAPTURE:show_version_all_nodes.txt
show version
_END

_WAIT: 120
_CAPTURE:BACKUP_CONFIGS/backup_config_10.100.100.3.txt
admin display-config
_END

_ERROR:(?i)(critical:|error:|minor:|major:)
configure
    port 1/4/8
        description "lab test"
exit all
